async function sendQueryToBackend(studentQuery) { const trimmedQuery = studentQuery.trim(); if (!trimmedQuery) return; statusEl.innerText = "余主任思考緊，準備回答你..."; if (currentAudio) { currentAudio.pause(); currentAudio = null; } chatHistory.push({ role: "user", content: trimmedQuery }); try { const response = await fetch('/api/ask', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ messages: chatHistory }) }); const data = await response.json(); if (!response.ok) { statusEl.innerText = `伺服器錯誤 (${response.status}): ${data.detail || data.error || '未知錯誤'}`; chatHistory.pop(); return; } const replyText = data.text || "余主任沒有回應。"; statusEl.innerText = replyText; if (data.text && !data.text.includes("連線失敗")) { chatHistory.push({ role: "assistant", content: replyText }); addQuestionToHistory(trimmedQuery, replyText); } else { chatHistory.pop(); } if (data.audio_url) { currentAudio = new Audio(data.audio_url); currentAudio.play().catch(e => { console.log("Autoplay blocked:", e); }); } } catch (error) { console.error("Fetch error:", error); statusEl.innerText = "網絡請求失敗：" + error.message; chatHistory.pop(); } }
